from dataclasses import dataclass, field
from ethereum.crypto.hashing import keccak256

# Canonical hash of empty code — used to identify EOA accounts
# SIMPLIFIED: Real Ethereum stores this in the account trie; we compute on access
EMPTY_CODE_HASH = keccak256(b"")

@dataclass
class Account:
    """Ethereum account state.

    Every address on Ethereum has an associated Account.
    EOAs (Externally Owned Accounts) have empty code.
    Contract accounts have deployed bytecode in the code field.

    # SIMPLIFIED: Storage is a plain dict, not a Merkle Patricia Trie.
    # Real Ethereum computes a storage_root hash from a storage trie.
    """
    nonce: int = 0
    balance: int = 0
    code: bytes = b""
    storage: dict[bytes, int] = field(default_factory=dict)

    @property
    def code_hash(self) -> bytes:
        """Keccak-256 hash of the account's code.
        # SIMPLIFIED: Computed on access. Real Ethereum stores codeHash in the trie.
        """
        return keccak256(self.code)

    @property
    def is_empty(self) -> bool:
        """EIP-161: An account is empty when it has no code, zero nonce, and zero balance."""
        return self.nonce == 0 and self.balance == 0 and self.code == b""

EMPTY_ACCOUNT = Account()
```

from ethereum.state.account import Account, EMPTY_ACCOUNT
from ethereum.crypto.hashing import keccak256
from ethereum.encoding.rlp import rlp_encode, int_to_rlp_bytes

class WorldState:
    """Dict-backed world state mapping addresses to accounts.

    # SIMPLIFIED: Uses a plain dict instead of Merkle Patricia Trie.
    # Real Ethereum computes the state root from the trie structure.
    # Our state_root is a deterministic hash of sorted account data.
    """

    def __init__(self):
        self._accounts: dict[bytes, Account] = {}

    def get_account(self, address: bytes) -> Account:
        """Return account at address, or EMPTY_ACCOUNT if not found."""
        return self._accounts.get(address, EMPTY_ACCOUNT)

    def set_account(self, address: bytes, account: Account) -> None:
        """Set account at address."""
        self._accounts[address] = account

    def has_account(self, address: bytes) -> bool:
        """Check if address has an explicit account in state."""
        return address in self._accounts

    def modify_balance(self, address: bytes, delta: int) -> None:
        """Add delta to account balance (delta can be negative)."""
        account = self.get_account(address)
        # Create a new account with updated balance to avoid mutating EMPTY_ACCOUNT
        updated = Account(
            nonce=account.nonce,
            balance=account.balance + delta,
            code=account.code,
            storage=dict(account.storage),
        )
        self.set_account(address, updated)

    def increment_nonce(self, address: bytes) -> None:
        """Increment account nonce by 1."""
        account = self.get_account(address)
        updated = Account(
            nonce=account.nonce + 1,
            balance=account.balance,
            code=account.code,
            storage=dict(account.storage),
        )
        self.set_account(address, updated)

    @property
    def state_root(self) -> bytes:
        """Compute a deterministic placeholder state root.

        # SIMPLIFIED: Real Ethereum computes the state root from a Merkle Patricia Trie.
        # We hash sorted account data to produce a deterministic but non-standard root.
        """
        if not self._accounts:
            return keccak256(b"")

        # Sort by address for determinism, encode each account's data
        items = []
        for address in sorted(self._accounts.keys()):
            account = self._accounts[address]
            account_data = rlp_encode([
                address,
                int_to_rlp_bytes(account.nonce),
                int_to_rlp_bytes(account.balance),
                account.code,
            ])
            items.append(account_data)

        return keccak256(rlp_encode(items))


def create_genesis_state(alloc: dict[bytes, int]) -> WorldState:
    """Create genesis world state with pre-funded accounts.

    Args:
        alloc: Mapping of address (20-byte bytes) to initial balance (wei).

    Returns:
        WorldState with accounts initialized at zero nonce, empty code, empty storage.
    """
    state = WorldState()
    for address, balance in alloc.items():
        account = Account(nonce=0, balance=balance, code=b"", storage={})
        state.set_account(address, account)
    return state
```
