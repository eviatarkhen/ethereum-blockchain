```python
def apply_transaction(tx, world_state, block_gas_limit):
    """Apply a signed transaction to world state.

    This is the core state transition function, following the Yellow Paper:
    1. Validate transaction (nonce, balance, gas)
    2. Snapshot state for potential rollback
    3. Deduct upfront gas cost from sender
    4. Increment sender nonce
    5. Transfer value (if applicable)
    6. Execute code (if recipient has code)
    7. On failure: revert state to snapshot (but keep nonce increment + gas consumption)
    8. Refund unused gas to sender

    # SIMPLIFIED: No gas refund mechanism for SSTORE clears.
    # SIMPLIFIED: No coinbase/validator reward.
    # SIMPLIFIED: No EIP-1559 base fee burning.
    """
    # Step 1: Validate
    sender = validate_transaction(tx, world_state, block_gas_limit)

    # Step 2: Snapshot
    state_snapshot = world_state.snapshot()

    try:
        # Step 3: Deduct upfront gas
        upfront_cost = tx.gas_limit * tx.gas_price
        world_state.deduct_balance(sender, upfront_cost)

        # Step 4: Increment nonce
        world_state.increment_nonce(sender)

        # Step 5 + 6: Execute
        gas_available = tx.gas_limit - calculate_intrinsic_gas(tx.data)

        if tx.to is None:
            # Contract creation -- placeholder for Plan 03-03
            result = _execute_create(tx, sender, world_state, gas_available)
        elif world_state.get_account(tx.to).code:
            # Contract call
            result = _execute_contract_call(tx, sender, world_state, gas_available)
        else:
            # Simple value transfer
            result = _execute_value_transfer(tx, sender, world_state, gas_available)

        if not result.success:
            # Revert everything except nonce increment and gas consumption
            world_state.revert(state_snapshot)
            world_state.deduct_balance(sender, upfront_cost)
            world_state.increment_nonce(sender)

        # Step 8: Refund unused gas
        gas_refund = result.gas_remaining * tx.gas_price
        world_state.add_balance(sender, gas_refund)

        return TransactionResult(
            success=result.success,
            return_data=result.return_data,
            gas_used=tx.gas_limit - result.gas_remaining,
            gas_remaining=result.gas_remaining,
            error=None if result.success else 'execution failed',
        )

    except Exception as e:
        # Catastrophic failure -- revert everything, consume all gas
        world_state.revert(state_snapshot)
        world_state.deduct_balance(sender, upfront_cost)
        world_state.increment_nonce(sender)
        return TransactionResult(
            success=False, return_data=b'',
            gas_used=tx.gas_limit, gas_remaining=0,
            error=str(e),
        )
```
